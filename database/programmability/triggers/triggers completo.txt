CREATE TABLE Bitacora (
    idBitacora SERIAL PRIMARY KEY,
    tabla VARCHAR(100),
    accion VARCHAR(10),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    usuario_bd VARCHAR(100),
    datos_antes TEXT,
    datos_despues TEXT
);

create trigger tg_bitacora_miembro
after insert or update or delete on Miembro
for each row
